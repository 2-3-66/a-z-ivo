# A-Z-IVO ENGINE V1 - Mureka-like MusiCoT # pip install audiocraft transformers torch from audiocraft.models import MusicGen import torch class AZivoEngine: def __init__(self): print("Chargement moteur A-Z-IVO...") # Moteur audio pro (comme Mureka utilise) self.model = MusicGen.get_pretrained('facebook/musicgen-medium') self.model.set_generation_params(duration=30) # 30 sec par bloc def musicot_plan(self, prompt, style): # ETAPE 1 - Structure-First (MusiCoT) # C'est le cerveau de Mureka style_map = { "coupe": "Coupé-décalé 130BPM, log drum percutant, Atalaku, festif", "zouglou": "Zouglou 95BPM, guitare mélancolique, chœur", "rap": "Rap Ivoire 140BPM, drill ivoire, kick lourd" } base = style_map.get(style, style) plan = f""" [Intro: 4 mesures, {base}] [Couplet 1: 8 mesures, voix homme nouchi, thème: {prompt}] [Refrain: 8 mesures, chœur, montée énergie, slogan A-Z-IVO] [Couplet 2: 8 mesures, flow rapide] [Outro: 4 mesures, fade log drum] """ return plan, f"{base}, {prompt}, high quality, Ivorian, Abidjan" def generate(self, prompt, style="coupe"): plan, audio_prompt = self.musicot_plan(prompt, style) print(f"PLAN MusiCoT:\n{plan}") print(f"Génération audio: {audio_prompt}") # ETAPE 2 - Generation-Second wav = self.model.generate([audio_prompt]) # wav est un tensor audio return plan, wav[0] # UTILISATION if __name__ == "__main__": engine = AZivoEngine() plan, audio = engine.generate("Yopougon nuit chaude, boss du game", "coupe") # Sauvegarde from audiocraft.data.audio import audio_write audio_write('azivo_output', audio.cpu(), engine.model.sample_rate, strategy="loudness") print("Fait: azivo_output.wav")
